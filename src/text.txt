Процедура СоздатьОплатуБезналичныйНаСервере()
    Для Каждого Строка Из ТаблицаДокументовЧеки Цикл
        
        Если НЕ Строка.Выбор Тогда
            Продолжить;
        КонецЕсли;
        
        Запрос = Новый Запрос;
        Запрос.Текст = 
        "ВЫБРАТЬ
        |    ЧекККМ.Ссылка       КАК Ссылка,
        |    ЧекККМ.СуммаДокумента КАК СуммаДокумента
        |ИЗ
        |    Документ.ЧекККМ КАК ЧекККМ
        |ГДЕ
        |    ЧекККМ.Дата = &Дата
        |    И ЧекККМ.Номер = &Номер";
        Запрос.УстановитьПараметр("Дата", Строка.Дата);
        Запрос.УстановитьПараметр("Номер", Строка.Номер);
        
        РезЗапроса = Запрос.Выполнить().Выбрать();
        Если НЕ РезЗапроса.Следующий() Тогда
            Продолжить;
        КонецЕсли;
        
        // Получаем объект и добавляем строку в табличную часть Оплата
        ОбъектЧек = РезЗапроса.Ссылка.ПолучитьОбъект();
        
        НоваяСтрока = ОбъектЧек.Оплата.Добавить();
        НоваяСтрока.ВидОплаты = ВыбранныйДоговорЭквайринга;
        НоваяСтрока.Сумма     = РезЗапроса.СуммаДокумента;
        
        ОбъектЧек.Записать(РежимЗаписиДокумента.Проведение)
        
    КонецЦикла;

    // Удаляем только выбранные строки

    Пока НомерСтроки >= 0 Цикл

&НаСервере
Процедура СравнитьИВывестиНаСервере(АдресФайла)
    
    ДвоичныеДанные = ПолучитьИзВременногоХранилища(АдресФайла);
    ИмяВременногоФайла = ПолучитьИмяВременногоФайла("xlsx");
    ДвоичныеДанные.Записать(ИмяВременногоФайла);
    
    ТабДок = Новый ТабличныйДокумент;
    ТабДок.Прочитать(ИмяВременногоФайла, СпособЧтенияЗначенийТабличногоДокумента.Значение);
    
    // Собираем номера накладных из колонки 3 начиная с 3 строки
    МассивНомеров = Новый Массив;
    НомерСтроки = 3;
    Пока НомерСтроки <= ТабДок.ВысотаТаблицы Цикл
        ЗначениеЯчейки = СокрЛП(ТабДок.Область(НомерСтроки, 3, НомерСтроки, 3).Текст);
        Если ЗначениеЗаполнено(ЗначениеЯчейки) Тогда
            МассивНомеров.Добавить(ЗначениеЯчейки);
        КонецЕсли;
        НомерСтроки = НомерСтроки + 1;
    КонецЦикла;
    
    УдалитьФайлы(ИмяВременногоФайла);
    
    Если МассивНомеров.Количество() = 0 Тогда
        Сообщить("В файле не найдены номера накладных!");
        Возврат;
    КонецЕсли;
	
	
	// Удаляем из таблицы строки которых нет в Excel
    НомерСтроки = ТаблицаДокументовЧеки.Количество() - 1;
    Пока НомерСтроки >= 0 Цикл
        Строка = ТаблицаДокументовЧеки[НомерСтроки];
        Если МассивНомеров.Найти(СокрЛП(Строка.Комментарий)) = Неопределено Тогда
            ТаблицаДокументовЧеки.Удалить(Строка);
        КонецЕсли;
        НомерСтроки = НомерСтроки - 1;
    КонецЦикла;
    
    Если ТаблицаДокументовЧеки.Количество() = 0 Тогда
        Сообщить("Совпадений не найдено! Проверьте файл Excel или период.");
    Иначе
        Сообщить("Готово! Показаны только совпавшие чеки.");
    КонецЕсли;
